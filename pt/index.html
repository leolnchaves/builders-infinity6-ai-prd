EXEMPLO RÁPIDO

Da ideia à primeira execução

Carregue o bundle do SDK, garanta a sessão e envie um arquivo com poucas linhas.

index.html
<script src="https://cdn.infinity6.ai/releases/sdkweb/latest/bundle/i6sdk-web.js"></script>
<input type="file" id="arquivo" />
<script type="module">
  async function main() {
    const sdk = new window.I6Sdk();
    // Sem sessão: redireciona para o login
    const user = await sdk.auth().requireUser();
    if (!user) return;

    const input = document.querySelector("#arquivo");
    input.addEventListener("change", async () => {
      const file = input.files[0];
      if (!file) return;
      await sdk.ingest().upload({
        dataset: "<seu-dataset>",
        table: "<sua-tabela>",
        partitions: { "<sua-chave>": "<seu-valor>" },
        file,
      });
    });
  }
  main();
</script>
Roda no navegador, carregado via tag script
