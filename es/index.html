EJEMPLO RÁPIDO

De la idea a la primera ejecución

Carga el bundle del SDK, asegura la sesión y sube un archivo en pocas líneas.

index.html
<script src="https://cdn.infinity6.ai/releases/sdkweb/latest/bundle/i6sdk-web.js"></script>
<input type="file" id="arquivo" />
<script type="module">
  async function main() {
    const sdk = new window.I6Sdk();
    // Sin sesión: redirige al login
    const user = await sdk.auth().requireUser();
    if (!user) return;

    const input = document.querySelector("#arquivo");
    input.addEventListener("change", async () => {
      const file = input.files[0];
      if (!file) return;
      await sdk.ingest().upload({
        dataset: "<tu-dataset>",
        table: "<tu-tabla>",
        partitions: { "<tu-clave>": "<tu-valor>" },
        file,
      });
    });
  }
  main();
</script>
Se ejecuta en el navegador, cargado mediante etiqueta script
